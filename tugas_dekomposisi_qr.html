Kode Verifikasi Python (NumPy)

import numpy as np

# Matriks A
A = np.array([
    [12, 5, 8, 3],
    [7, 14, 1, 9],
    [6, 2, 11, 4],
    [10, 13, 15, 0]
], dtype=float)

# QR Decomposition (Gram-Schmidt)
Q, R = np.linalg.qr(A)

print("Matriks A:")
print(A)

print("\nMatriks Q:")
print(Q)

print("\nMatriks R:")
print(R)
